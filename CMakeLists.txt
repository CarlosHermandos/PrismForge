cmake_minimum_required(VERSION 3.22)
project(PrismForge VERSION 0.3.1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_OSX_DEPLOYMENT_TARGET 12.0 CACHE STRING "Minimum macOS")
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/ThirdParty/JUCE/CMakeLists.txt")
 set(PRISM_DEFAULT_JUCE "${CMAKE_CURRENT_SOURCE_DIR}/ThirdParty/JUCE")
else()
 set(PRISM_DEFAULT_JUCE "$ENV{HOME}/Documents/JUCE")
endif()
set(JUCE_PATH "${PRISM_DEFAULT_JUCE}" CACHE PATH "JUCE source")
set(PRISM_MANUFACTURER "Prism Forge Audio" CACHE STRING "Manufacturer")
set(PRISM_BUNDLE_ID "audio.prismforge.instrument" CACHE STRING "Bundle ID")
set(PRISM_MANUFACTURER_CODE PrFg CACHE STRING "Four-character manufacturer code")
set(PRISM_PLUGIN_CODE PrF1 CACHE STRING "Four-character plugin code")
add_subdirectory("${JUCE_PATH}" "${CMAKE_BINARY_DIR}/JUCE")
juce_add_plugin(PrismForge COMPANY_NAME "${PRISM_MANUFACTURER}" BUNDLE_ID "${PRISM_BUNDLE_ID}"
 PLUGIN_MANUFACTURER_CODE ${PRISM_MANUFACTURER_CODE} PLUGIN_CODE ${PRISM_PLUGIN_CODE}
 FORMATS AU VST3 Standalone PRODUCT_NAME "Prism Forge" IS_SYNTH TRUE NEEDS_MIDI_INPUT TRUE
 NEEDS_MIDI_OUTPUT FALSE COPY_PLUGIN_AFTER_BUILD FALSE AU_MAIN_TYPE kAudioUnitType_MusicDevice)
juce_add_binary_data(PrismLegal NAMESPACE PrismLegalData SOURCES LICENSE.md NOTICE.md)
target_link_libraries(PrismForge PRIVATE PrismLegal)
target_sources(PrismForge PRIVATE Source/Plugin.cpp Source/Editor.cpp Source/PresetBrowser.cpp Source/AssetIO.cpp Source/WavetableEditor.cpp)
target_compile_definitions(PrismForge PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(PrismForge PRIVATE juce::juce_audio_utils PUBLIC juce::juce_recommended_config_flags)
include(CTest)
add_executable(PrismCoreTests Tests/CoreTests.cpp)
target_include_directories(PrismCoreTests PRIVATE Source)
add_test(NAME Core COMMAND PrismCoreTests "${CMAKE_SOURCE_DIR}/renders")
add_executable(PrismProcessorTests Tests/ProcessorTests.cpp)
target_link_libraries(PrismProcessorTests PRIVATE PrismForge)
target_include_directories(PrismProcessorTests PRIVATE Source)
add_test(NAME Processor COMMAND PrismProcessorTests)
if(APPLE)
 add_executable(PrismAUSmoke Tests/AUSmoke.cpp)
 target_include_directories(PrismAUSmoke PRIVATE Source)
 target_link_libraries(PrismAUSmoke PRIVATE "-framework AudioToolbox" "-framework CoreFoundation")
 add_test(NAME AUIntegration COMMAND PrismAUSmoke "$<TARGET_BUNDLE_DIR:PrismForge_AU>")
endif()
target_include_directories(PrismProcessorTests PRIVATE "${JUCE_PATH}/modules")
target_compile_definitions(PrismProcessorTests PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1)
if(APPLE)
 foreach(format AU VST3 Standalone)
  add_custom_command(TARGET PrismForge_${format} POST_BUILD
   COMMAND /usr/bin/codesign --force --sign - "$<TARGET_BUNDLE_DIR:PrismForge_${format}>" VERBATIM)
 endforeach()
endif()
add_executable(PrismInstrumentTests Tests/InstrumentTests.cpp)
target_include_directories(PrismInstrumentTests PRIVATE Source)
add_test(NAME Instrument COMMAND PrismInstrumentTests)
add_executable(PrismAssetTests Tests/AssetTests.cpp)
target_include_directories(PrismAssetTests PRIVATE Source "${JUCE_PATH}/modules")
target_compile_definitions(PrismAssetTests PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1)
target_link_libraries(PrismAssetTests PRIVATE PrismForge)
add_test(NAME Assets COMMAND PrismAssetTests)
add_executable(PrismHybridTests Tests/HybridTests.cpp)
target_include_directories(PrismHybridTests PRIVATE Source "${JUCE_PATH}/modules")
target_compile_definitions(PrismHybridTests PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1)
target_link_libraries(PrismHybridTests PRIVATE PrismForge)
add_test(NAME Hybrid COMMAND PrismHybridTests)
add_executable(PrismSequencerTests Tests/SequencerTests.cpp)
target_include_directories(PrismSequencerTests PRIVATE Source)
add_test(NAME Sequencer COMMAND PrismSequencerTests)
add_executable(PrismBenchmark Tests/Benchmark.cpp)
target_include_directories(PrismBenchmark PRIVATE Source)
add_executable(PrismContentGenerator Tests/ContentGenerator.cpp)
target_include_directories(PrismContentGenerator PRIVATE Source "${JUCE_PATH}/modules")
target_compile_definitions(PrismContentGenerator PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1)
target_link_libraries(PrismContentGenerator PRIVATE PrismForge)
if(APPLE)
 add_executable(PrismVST3Smoke Tests/VST3Smoke.cpp)
 target_include_directories(PrismVST3Smoke PRIVATE "${JUCE_PATH}/modules/juce_audio_processors_headless/format_types/VST3_SDK")
 target_link_libraries(PrismVST3Smoke PRIVATE "-framework CoreFoundation")
 add_test(NAME VST3Integration COMMAND PrismVST3Smoke "$<TARGET_BUNDLE_DIR:PrismForge_VST3>")
endif()
